namespace pdm
{
    void Core::addModule(std::unique_ptr<shared::IModule> module)
    {
        // Получаем ID модуля (копируем строку, так как модуль может быть удален)
        auto id(module->getId());

        // Проверяем, есть ли уже модуль с таким ID
        if(m_modules.contains(id))
        {
            std::cerr << std::format("[Core] Warning: Module with ID [{}] is already registered!\n",
                                    shared::idToString(id));
            return;
        }
        std::cout << std::format("[Core] Registered module: {} (v{}).\n",
                                 module->getName(), module->getVersion());

        m_modules.emplace(id, std::move(module));
    }

    void Core::runModule(std::string_view moduleName, const shared::JobParams& params)
    {
        auto opId{ shared::ParseModuleId(moduleName) };
        if(!opId)
        {
            std::cerr << std::format("[Core] Error: Module with name [{}] not found.\n", moduleName);
            return;
        }
        auto it{ m_modules.find(*opId) };

        if(it == m_modules.end())
        {
            std::cerr << std::format("[Core] Error: Module with ID [{}] not found.\n", shared::idToString(*opId));
            return;
        }

        std::cout << std::format("[Core] Run module with ID: [{}].\n", shared::idToString(*opId));

        it->second->execute(params);
    }
}

    /**
     * @brief Ядро системы (оркестратор).
     *
     * Отвечает за регистрацию модулей и их запуск. Ядро ничего не знает
     * о внутреннем устройстве конкретных модулей (rename, ocr и т.д.),
     * оно работает только через интерфейс IModule.
     * Это обеспечивает слабую связанность (loose coupling) и позволяет
     * легко расширять систему новыми модулями.

    class Core final
    {

         * @brief Реестр (хранилище) всех зарегистрированных модулей.
         *
         * Используется std::unordered_map для быстрого поиска модуля по его ID
         * за константное время O(1). Ключом является enum class ID модуля,
         * значением — умный указатель unique_ptr, который владеет модулем.
         */
        std::unordered_map<shared::ModuleId, std::unique_ptr<shared::IModule>> m_modules;

    public:
        Core() = default;
        ~Core() = default;

        /**
         * @brief Зарегистрировать новый модуль в ядре.

         * Если модуль с таким же ID уже зарегистрирован, выводится предупреждение
         * и новый модуль не добавляется (защита от дубликатов).
         *
         * @param module Умный указатель на модуль, реализующий интерфейс IModule.
         */
        void addModule(std::unique_ptr<shared::IModule> module);

         * @param moduleName Строковый идентификатор модуля (например, "rename").

         */

# PDM-система для конструкторского бюро

## 1. Назначение

Хочу автоматизировать работу конструкторского бюро.
Цель развить в полноценную PDM систему. На данном
этапе первоочередная цель - переименование 
отсканированных чертежей.

Сопутствующая цель - учеба и практика программирования

## 2. Слои

### apps
исполняемые файлы-запускалки. Минимум кода: `BuildCore()` + `runModule()`.
Сегодня — core-server (консоль). Позже — ui-desktop/ui-web.
Никакой бизнес-логики, никакой проводки зависимостей — только вызов composition.

### core
Ядро
Логика домена: правила работы с документами, хранение, регистрация модулей, общие сервисы (логирование, конфигурация).
Не знает о конкретных модулях — знает только интерфейс IModule.

### composition
composition root. Единственное место, где разрешено знать все конкретные
реализации (Rename, QxlsxReader). Собирает граф и возвращает готовый Core.
Не содержит бизнес-логики.

### modules
независимые модули.
Один модуль = одна функция (переименование, архивирование, OCR — по мере роста).
Не знают друг о друге. Регистрируются в core через IModule.

### shared
общие типы и контракты.
* Типы домена: Document, Designation, File, Revision.
* Интерфейсы: IModule, ILogger, IRepository.
* Общие константы и enum'ы.
Не содержит логики. Не зависит ни от кого.

### third_party
заимствованные библиотеки для работы программы, изолированный слой. отключены предупреждения

### tests
unit тесты для проверки работы с целью контроля состояния.
Линкуются с core и modules, чтобы проверять их работу.
Не входят в состав продукта. Используют Catch2 из third_party.


## 3. Правило зависимостей

composition (composition root)
   ├─→ core → shared
   ├─→ modules → shared
   └─→ infrastructure → shared

core-server (exe)
   └─→ composition

**composition** — единственное место, где разрешено знать все конкретные
реализации (Rename, QxlsxReader). Собирает граф объектов и возвращает
готовый `std::unique_ptr<pdm::Core>`. Не содержит бизнес-логики.

**core-server** — тонкий запускатель. Знает только `composition`
(получить Core) и `shared`-типы (JobParams). Никаких `#include` из
`modules/` или `infrastructure/`.
core-server знает composition, shared-типы и контрактные
заголовки модулей (rename_keys.h). Запрещено включать 
rename.h, qxlsx_reader.h и любые заголовки с реализацией.

**core** не знает о конкретных модулях — только `IModule` и `ModuleId` из shared.

**shared** не знает ни о ком. Не содержит логики.

## 4. Принятые ограничения
* Язык: C++20.

* Компилятор предупреждения трактует как ошибки, кроме third_party.

* Целевая ОС: Windows 10/11, без прав администратора.

* Интернет на рабочей машине недоступен.

* Работа только в локальной сети.

* Хранение: пока без БД, потом SQLite → PostgreSQL.

* UI: пока консоль, потом Qt или браузер (не решено).

* Внешние библиотеки — через git submodule в third_party.

## 5. Что не входит на первом этапе

* OCR и распознавание.

* Обезличивание КД.

* Карточки документов.

* Применяемость, проекты.

* Многопользовательский режим.

* Какой UI — Qt или браузер?

## Интерфейс IModule

Любой модуль системы должен реализовывать этот интерфейс, чтобы ядро (core) могло им управлять.

**1. Свойства модуля:**
*   `getId()` — типизированный идентификатор `ModuleId` (enum class).
*   `getName()` — человекочитаемое имя (например, "Переименование сканов").
*   `getVersion()` — версия модуля (например, "1.0.0").

Почему enum, а не строка: набор встроенных модулей закрыт и известен
на этапе компиляции. Enum даёт проверку опечаток компилятором, `-Wswitch`
ловит «забыл обработать новый модуль», нет аллокаций на find/сравнение.
Строки остаются только на границе — ParseModuleId/idToString в
module_id.h. Если завтра появятся динамические плагины (.dll без
пересборки ядра), ключом станет строковый/UUID id — но это будет
осознанное решение под реальное требование, а не «на всякий случай»

**2. Выполнение задачи:**
*   `execute(Параметры)` — главный метод. Ядро передает ему параметры (пути к файлам/папкам). 
*   Модуль возвращает статус (Успешно / Ошибка) и текст сообщения (для лога).

**3. Параметры для модуля rename:**
*   `excelPath` — путь к файлу спецификации.
*   `scansFolder` — путь к папке со сканами.
## Модель Document

Документ однозначно определяется тремя полями: **Номер**, **Код**, **Наименование**.

| Поле | Тип | Обязательность |
|---|---|---|
| Номер | текст | обязательно |
| Код | enum / текст | опционально |
| Наименование | текст | обязательно |

Уникальность — по паре (Номер, Код). Если код не указан, это считается отдельным значением ключа. Вопрос о конфликте «чертёж детали без кода» и «спецификация без кода» — открытый

## Справка: поля документа по ГОСТ (не модель, на будущее)
1. Наименование (текст) - обязательное, наименование изделия (прим. Балка) и документа (прим. Сборочный чертеж);
2. Обозначение (текст) - обязательное, обозначение (прим. 511.01.01.101) и код (прим. СБ, ТУ и т.п.) уникальное;
* групповой чертеж (bool) да нет - подтягивание основного документа
3. Обозначение материала (текст) - обязателно только в деталях;
4. Масса изделия (double) - необязательно, но флаг что нет массы;
5. Масштаб (текст) - необязательно;
6. Порядковый номер листа (int) - обязательно;
7. Общее количество листов документа (int) - необязательно;
8. Наименование или код организации (текст) - необязательно;
9. Сведения оподписании документа (текст) - необязательно;
* возможно несколько полей - разраб., пров., утв. и т.д.
10. Сведения об изменениях документа - обязательно
* Порядковый номер изменения (int) - обязательно
* Номер документа на основании которого произведено изменение (текст) - необязательно.
11. инвентарный номер подлинника (текст) - необязательно
12. Сведения о приемке подлинника в службу технической документации (СТД) (текст) - необязательно
13. Инвентарный номер подлинника, взамен которого выпущен данный подлинник (текст) - необязательно;
14. Инвентарный номер дубликата (текст) - необязательно;
15. Сведения о приемке дубликата в СТД (текст) - необязательно;
16.  Обозначение документа, взамен или на основании которого выпущен данный документ (текст) - необязательно;
17. Обозначение соответствующего документа, в котором впервые записан данный документ (текст) - необязательно;
18. Знак, установленный заказчиком (bool) - да/нет необязательно (по умолчанию нет)
19. Литера (текст) - необязательно;
20. Сведения об утверждении документации соответствующей литеры (текст) - необязательно;
21. Сведения об утверждении документации (текст) - необязательно;
22. Индекс заказчика (текст) - необязательно;
23. Обозначение формата листа (текст) - необязательно;
24. Версия документа (текст) - необязательно;
25. Уровень конфиденциальности документа (текст) - необязательно;
26. Ссылка файла документа (текст) - обязательно;
27. Ссылки на файл компас (текст) - необязательно.
# Модуль переименования сканов (rename)

## 1. Назначение
Модуль переименовывает отсканированные файлы из формата data001.jpg
в новое имя в формате (обозначение)(код) (наименование).jpg

## 2. Входные данные
Получает файл спецификации в формате .xls и папку со сканами,
в которой сложены файлы *.jpg

## 3. Выходные данные
Лог переименования в формате CSV (было;стало).
Переименованные файлы (обозначение)(код) (наименование).jpg
Data001.jpg -> 123.45.67.890СБ Форма.jpg.
Код пишется слитно с номером.

## 4. Алгоритм (шаги)
1. Открыть Excel.
2. Пройти по первому листу (Спецификация А4) и извлечь список документов:
    * Отсечение «Стандартных изделий». Программа читает строки сверху вниз.
    Как только она встречает в колонке «Наименование» строку 
    «Стандартные изделия», она полностью прекращает чтение этого листа.
    * Пустые строки и заголовки. Если в строке пусто в колонке «Обозначение»
    или «Наименование» (например, шапки таблиц, пустые строки-разделители)
    — программа их пропускает.
    * Документы без чертежей (БЧ). Если в колонке «Примечание» (столбец V)
    стоит «БЧ» программа пропускает эту строку (так как переименовывать 
    нечего, скана нет).
3. Найти в папке все файлы data*.jpg и отсортировать их по номеру.
4. Проверка количества: Сравнить, сколько файлов в папке и сколько 
документов осталось в списке после фильтрации.
    * Если не совпадает — СТОП (вывести ошибку).
5. Если совпадает — сформировать имена файлов по шаблону
<Номер><Код (если есть)> <Наименование>.jpg и переименовать 
(с логом и проверкой на дубли).
    

## 5. Правила и ограничения
1. Пропускаются строки:
    * Пустые в столбце C.
    * Имеющие "БЧ" в столбце V.
    * Строка в столбце G == "Стандартные изделия" и все что ниже.
2. Количество строк == количеству файлов папке со сканами (временно, при прототипировании) 
3. Работа только со спецификацией А4

## 6. Обработка ошибок
1. **Файл Excel не найден или занят:** Вывести понятное сообщение,
не закрывать программу аварийно.
2. **Несовпадение количества:** Если файлов в папке больше или меньше,
чем документов в спецификации — вывести ошибку (например: "В папке
найдено 15 файлов, в спецификации 14 строк. Переименование отменено.").
3. **Ошибка доступа к файлу (нет прав / файл открыт в другой программе):**
Записать ошибку в лог, перейти к следующему файлу, не прерывать весь процесс.
4. **Дубликаты имен:** Если программа сгенерировала два одинаковых имени
(например, два раза `123 Планка.jpg`) — остановиться, вывести ошибку.

## 7. Нерешенные вопросы
1. В спецификации .xls не учтены сборки т.е. под номером
123.45.67.890 скрывается отсканированные 
спецификация 123.45.67.890.jpg, и сборка 123.45.67.890СБ.jpg (но не обязательно - 
сборки как документа может не быть, но в 99 % сборка есть);
2. В спецификации .xls не учтено количество листов (например для документа в спецификации
.xls будет одна запись 123.45.67.890, а отсканированных файлов может быть 
data001.jpg - 123.45.67.890 лист 1, data002.jpg - 123.45.67.890 лист 2,
data003.jpg - 123.45.67.890СБ лист 1, data004.jpg - 123.45.67.890СБ лист 2).

* Тип документа.

* Уникальность при пустом коде.

    class QxlsxReader final : public pdm::shared::IExcelReader
    {
    private:
        std::unique_ptr<QXlsx::Document> m_xlsx{};

    public:
        QxlsxReader() = default;

        explicit QxlsxReader(const std::string& filePath)
        {
            open(filePath);
        }

        ~QxlsxReader() override = default;

        bool open(const std::string& filePath) override
        {
            m_xlsx = std::make_unique<QXlsx::Document>( QString::fromStdString(filePath) );

            return m_xlsx->load();
        }

        /**
         * @brief Прочитать список документов из открытого файла Excel.
         *
         * @details
         * Метод проходит по строкам листа сверху вниз, используя реальный размер таблицы (dimension().lastRow()).
         * Для каждой строки проверяются условия:
         * 1. Если в столбце G (Наименование) встречается "Стандартные изделия" или "Прочие изделия",
         *    чтение немедленно прекращается (всё, что ниже — болты и гайки, их не переименовываем).
         * 2. Если строка пустая (нет ни обозначения, ни наименования), она пропускается.
         * 3. Если в столбце V (Примечание) стоит "БЧ", строка пропускается (нет скана).
         * 4. Если в строке есть и Обозначение, и Наименование — она добавляется в список.
         *
         * @note
         * Номера столбцов (C=3, G=7, V=22) жестко зафиксированы под шаблон
         * спецификации A4, утвержденный на предприятии.
         *
         * @return Вектор структур Document с извлеченными данными.
         */
        std::vector<pdm::shared::Document> readDocuments() override
        {
            std::vector<pdm::shared::Document> documents;

            const int lastRow{ m_xlsx->dimension().lastRow() };

            // Если таблица пустая, сразу выходим
            if (lastRow <= 0)
            {
                return documents;
            }

            for(int row{ 1 }; row <= lastRow; ++row)
            {
                // Читаем значения из нужных ячеек.
                // Столбцы: C=3 (Обозначение), G=7 (Наименование), V=22 (Примечание)
                QString cellC{ m_xlsx->read(row, 3).toString().trimmed() };
                QString cellG{ m_xlsx->read(row, 7).toString().trimmed() };
                QString cellV{ m_xlsx->read(row, 22).toString().trimmed() };

                // Если достигли раздела "Стандартные изделия" — останавливаемся
                if(cellC.isEmpty() &&
                                     (cellG.contains("Стандартные изделия", Qt::CaseInsensitive) ||
                                      cellG.contains("Прочие изделия", Qt::CaseInsensitive) ||
                                      cellG.contains("Материалы", Qt::CaseInsensitive)))
                {
                    break;
                }

                // Пропускаем пустые строки и строки-заголовки
                if(cellC.isEmpty() && cellG.isEmpty())
                {
                    continue;
                }

                // Пропускаем документы без чертежей (БЧ)
                if(cellV.contains("БЧ", Qt::CaseInsensitive))
                {
                    continue;
                }

                // Если есть и номер, и наименование — это валидный документ
                if(!cellC.isEmpty() && !cellG.isEmpty())
                {
                    pdm::shared::Document doc;
                    doc.number = cellC.toStdString();
                    doc.name = cellG.toStdString();

                    documents.emplace_back(std::move(doc));
                }
            }
            // Читаем ячейку C2 (строка 2, колонка 3 — в QXlsx колонки нумеруются с 1)

            return documents;
        }
    };

        constexpr float kMaxImageDimension{ 1024.0f };
        constexpr int kJpegQuality{ 90 };

        struct StbiImageDeleter
        {
            void operator()(unsigned char* img) const
            {
                if(img)
                {
                    stbi_image_free(img);
                }
            }
        };

        using UniqueStbiImage = std::unique_ptr<unsigned char, StbiImageDeleter>;

        /**
         * @brief Кодирование массива байтов в Base64.
         * @param data Взгляд на массив байтов (std::span).
         * @return Строка в формате Base64.
         */
        std::string base64Encode(std::span<const unsigned char> data)
        {
            static constexpr std::string_view base64Chars{ "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/" };

            constexpr int kBitsPerChar{ 6 };
            constexpr int kBitsPerByte{ 8 };
            constexpr int kMask{ 0x3F }; // 6 бит (111111)
            constexpr int kBlockSize{ 4 };

            std::string out{};
            out.reserve((data.size() + 2) / 3 * kBlockSize );

            int val{ 0 };
            int valb{ -kBitsPerChar };
            for(const auto& c : data)
            {
                val = (val << kBitsPerByte) + c;
                valb += kBitsPerByte;
                while(valb >= 0)
                {
                    out.push_back(base64Chars[(val >> valb) & kMask]);
                    valb -= kBitsPerChar;
                }
            }

            if(valb > -kBitsPerChar)
            {
                out.push_back(base64Chars[((val << kBitsPerByte) >> (valb + kBitsPerByte)) & kMask]);
            }

            while(out.size() % kBlockSize)
            {
                out.push_back('=');
            }
            return out;
        }



        /**
         * @brief Callback для stb_image_write, чтобы сохранить JPEG в память (в vector), а не в файл.
         */
        void writeToVector(void* context, void* data, int size)
        {
            auto* vec{ static_cast<std::vector<unsigned char>*>(context) };
            auto* bytes{ static_cast<unsigned char*>(data) };
            vec->insert(vec->end(), bytes, bytes + size);
        }

std::string VLMStampRecognizer::encodeToBase64(const std::filesystem::path& imagePath)
{
    int width{};
    int height{};
    int channels{};

    UniqueStbiImage img{ stbi_load(imagePath.string().c_str(), &width, &height, &channels, 0) };

    if(!img) return "";

    // --- ТОЧНЫЙ КРОП (Только штамп) ---
    // Берем 35% ширины справа и 15% высоты снизу.
    int cropW = static_cast<int>(width * 0.35f);
    int cropH = static_cast<int>(height * 0.15f);
    int startX = width - cropW;
    int startY = height - cropH;

    std::vector<unsigned char> croppedImg(cropW * cropH * channels);

    for (int y = 0; y < cropH; ++y)
    {
        for (int x = 0; x < cropW; ++x)
        {
            for (int c = 0; c < channels; ++c)
            {
                croppedImg[(y * cropW + x) * channels + c] =
                    img.get()[((startY + y) * width + (startX + x)) * channels + c];
            }
        }
    }

    // Увеличили лимит до 2048, чтобы текст в штампе был крупным и четким
    int newWidth = cropW;
    int newHeight = cropH;
    const float maxSize = 2048.0f;

    if (cropW > maxSize || cropH > maxSize)
    {
        float scale = std::min(maxSize / cropW, maxSize / cropH);
        newWidth = static_cast<int>(cropW * scale);
        newHeight = static_cast<int>(cropH * scale);
    }

    std::vector<unsigned char> resizedImg(newWidth * newHeight * channels);
    auto layout = (channels == 4) ? STBIR_RGBA : STBIR_RGB;

    stbir_resize_uint8_linear(croppedImg.data(), cropW, cropH, 0,
                              resizedImg.data(), newWidth, newHeight, 0, layout);

    std::vector<unsigned char> jpegBuffer{};
    stbi_write_jpg_to_func(writeToVector, &jpegBuffer, newWidth, newHeight, channels, resizedImg.data(), kJpegQuality);

    return base64Encode(jpegBuffer);
}




std::optional<shared::Document> VLMStampRecognizer::recognize([[maybe_unused]] const std::filesystem::path& imagePath)
{
    std::string base64Image{ encodeToBase64(imagePath) };
    if(base64Image.empty())
    {
        return std::nullopt;
    }

    std::string prompt =
        "Ты — OCR-система. Твоя задача — прочитать текст в штампе на картинке. "
        "Верни ТОЛЬКО JSON без markdown-разметки.\n"
        "{\n"
        "  \"number\": \"строка (только номер, без кода)\",\n"
        "  \"code\": \"строка (СБ, ТУ и т.д. Если нет, то пусто)\",\n"
        "  \"name\": \"строка (наименование)\",\n"
        "  \"sheet\": число,\n"
        "  \"totalSheets\": число\n"
        "}\n"
        "ВАЖНО: Не придумывай данные! Читай только то, что видишь. Если не видишь — оставь пустую строку.";

    // 3. Собираем JSON для отправки (формат OpenAI API, который понимает llama.cpp)
    nlohmann::json requestBody{
        {"model", "gpt-3.5-turbo"}, // Сервер игнорирует это поле, но требует его наличия
        {"temperature", 0.1},       // 0.1 делает модель максимально точной и убирает "фантазии"
        {"max_tokens", 512},
        {"messages", nlohmann::json::array({
            {
                {"role", "user"},
                {"content", nlohmann::json::array(
                    {
                    {{"type", "text"}, {"text", prompt}},
                    {{"type", "image_url"}, {"image_url", {{"url", "data:image/jpeg;base64," + base64Image}}}}
                    }
                    )
                }
            }
        })}
    };

    httplib::Client cli(m_serverUrl);

    // Устанавливаем таймаут чтения: 300 секунд (5 минут).
    // Если нейросеть работает на CPU, она может думать долго.
    cli.set_read_timeout(300, 0);


    std::cout << "[VLM] Sending request to AI server for " << imagePath.filename().string() << "...\n";

    auto res = cli.Post("/v1/chat/completions", requestBody.dump(), "application/json");

    // 5. Проверяем, что запрос вообще дошел
    if (!res)
    {
        std::cerr << "[VLM] HTTP Error: Could not connect to " << m_serverUrl << "\n";
        return std::nullopt;
    }
    if (res->status != 200)
    {
        std::cerr << "[VLM] Server returned HTTP status " << res->status << "\n";
        return std::nullopt;
    }

    try
    {
        // 6. Разбираем ответ сервера
        auto responseJson = nlohmann::json::parse(res->body);
        std::string content = responseJson["choices"][0]["message"]["content"];

        std::cout << "[VLM] RAW AI RESPONSE:\n" << content << "\n-----------------------------------\n";

        // 7. Очищаем ответ от "мусора"
        // Нейросети часто оборачивают JSON в markdown (```json ... ```) или пишут "Вот результат:".
        // Нам нужно вырезать только текст от первой '{' до последней '}'.
        size_t start = content.find('{');
        size_t end = content.rfind('}');

        if (start != std::string::npos && end != std::string::npos && start < end)
        {
            content = content.substr(start, end - start + 1);
        }
        else
        {
            std::cerr << "[VLM] Error: AI did not return a valid JSON block.\n";
            std::cerr << "[VLM] Raw AI answer: " << content << "\n";
            return std::nullopt;
        }

        // 8. Парсим вырезанный JSON
        auto docJson = nlohmann::json::parse(content);

        shared::Document doc;
        doc.number = docJson.value("number", "");
        doc.name = docJson.value("name", "");

        // Безопасное получение целых чисел (ИИ может вернуть их как строку)
        auto getSafeInt = [](const nlohmann::json& j, const std::string& key, int defaultVal) {
            if (!j.contains(key)) return defaultVal;
            if (j[key].is_number()) return j[key].get<int>();
            if (j[key].is_string()) {
                try { return std::stoi(j[key].get<std::string>()); }
                catch (...) { return defaultVal; }
            }
            return defaultVal;
        };

        doc.sheet = getSafeInt(docJson, "sheet", 1);
        doc.totalSheets = getSafeInt(docJson, "totalSheets", 1);

        if (doc.number.empty() || doc.name.empty())
        {
            std::cerr << "[VLM] Error: AI returned empty number or name.\n";
            return std::nullopt;
        }

        // 9. Маппинг строкового кода (СБ, ТУ) в наш enum DocumentTypeCode
        std::string codeStr = docJson.value("code", "");

        // Приводим к верхнему регистру на всякий случай (Сб, сб, СБ -> СБ)
        for (char& c : codeStr) c = std::toupper(static_cast<unsigned char>(c));

        if (codeStr == "СБ") doc.code = shared::DocumentTypeCode::AssemblyDrawing;
        else if (codeStr == "ТУ") doc.code = shared::DocumentTypeCode::TechnicalSpecifications;
        else if (codeStr == "ВП") doc.code = shared::DocumentTypeCode::PurchasedPartsList;
        else doc.code = shared::DocumentTypeCode::Unknown;

        return doc;
    }
    catch (const std::exception& e)
    {
        std::cerr << "[VLM] JSON Parse Error: " << e.what() << "\n";
        std::cerr << "[VLM] Raw response body: " << res->body << "\n";
        return std::nullopt;
    }

}

std::string Rename::buildNewName(const shared::Document& doc, const std::filesystem::path& oldPath, int page)
{
    // ИСПРАВЛЕНО: Если листов больше 1, добавляем " лN", иначе оставляем пустым
    std::string pageSuffix{ (doc.totalSheets > 1) ? (" л" + std::to_string(page)) : "" };

    return std::format("{}{} {}{}{}", doc.number,
                       shared::DocumentTypeCodeToString(doc.code),
                       doc.name,
                       pageSuffix,
                       oldPath.extension().string());
}



void Rename::execute(const shared::JobParams& params)
{
    std::cout << "[RenameModule] Module runs and ready to work!\n";

    // Проверяем, передало ли ядро нужные пути
    if(!params.has(rename::keys::scansFolder))
    {
        std::cerr << "[RenameModule] Error: paths ways has not found!\n";
        return;
    }

    const auto& scansFolder{ params.get(rename::keys::scansFolder) };

    try
    {
        auto scanFiles{ getScansPathsFromFolder(scansFolder) };

        performRename(scanFiles, scansFolder);
    }
    catch(const std::exception& e)
    {
        std::cerr << e.what() << '\n';
        return;
    }

    std::cout << "[RenameModule] Module end work!\n";
}



std::vector<std::filesystem::path> Rename::getScansPathsFromFolder(std::filesystem::path folder)
{
    auto scanFiles{ getFilesFromFolder(folder, ".jpg", ".JPG", ".jpeg", ".JPEG") };

    std::ranges::sort(scanFiles);

    return scanFiles;
}



void Rename::performRename(const std::vector<std::filesystem::path>& scanFiles,
                           const std::filesystem::path& scansFolder)
{
    auto logPath{ scansFolder / "rename_log.csv" };
    RenameLogger logger{ logPath };

    RenameState state;

    for(const auto& file : scanFiles)
    {
        processSingleFile(file, state, logger);
    }

    std::cout << "\n=== Summary ===\n";
    std::cout << "Renamed: " << state.successCount << "\n";
    std::cout << "Errors:  " << state.errorCount << "\n";
    std::cout << "Log saved to: " << logPath << "\n";
}



void Rename::processSingleFile(const std::filesystem::path& file,
                               RenameState& state,
                               RenameLogger& logger)
{
    auto recognizedOpt{ m_recognizer->recognize(file) };

    if(!recognizedOpt.has_value())
    {
        std::string errorMsg{ "Cannot recognize stamp on file" };
        logger.log(file.filename().string(), "UNKNOWN", "ERROR: " + errorMsg);
        ++state.errorCount;

        // Сбрасываем состояние, так как последовательность прервалась
        state.previousDoc = std::nullopt;
        return;
    }

    const auto& doc{ recognizedOpt.value() };

    // Проверяем, является ли этот документ продолжением предыдущего
    if (doc.sheet > 0)
    {
        state.currentPage = doc.sheet;
    }
    else
    {
        if(state.previousDoc.has_value() &&
            state.previousDoc->number == doc.number &&
            state.previousDoc->code == doc.code)
        {
            // Это следующая страница того же документа
            ++state.currentPage;
        }
        else
        {
            // Это новый документ
            state.currentPage = 1;
        }
    }
    auto newPath{ file.parent_path() / buildNewName(doc, file, state.currentPage) };

    if(renameOneFile(file, newPath, logger))
    {
        ++state.successCount;
    }
    else
    {
        ++state.errorCount;
    }

    state.previousDoc = doc;
}



bool Rename::renameOneFile(const std::filesystem::path& oldPath, const std::filesystem::path& newPath, RenameLogger& logger)
{
    bool success{ false };

    try
    {
        // Проверяем, не существует ли уже файл с таким именем
        if(std::filesystem::exists(newPath))
        {
            std::string msg{ "Target file already exist" };
            logger.log(oldPath.filename().string(), newPath.filename().string(), "ERROR: " + msg);
        }
        else
        {
            std::filesystem::rename(oldPath, newPath);
            logger.log(oldPath.filename().string(), newPath.filename().string(), "OK");
            success = true;
        }
    }
    catch(const std::filesystem::filesystem_error& e)
    {
        logger.log(oldPath.filename().string(), newPath.filename().string(), std::string{ "ERROR: " } + e.what());
    }

    return success;
}

    /**
     * @brief Модуль переименования отсканированных чертежей.
     *
     * Читает спецификацию, анализирует файлы в папке и переименовывает их
     * в соответствии с обозначениями и наименованиями из спецификации.
     */
    class Rename final : public shared::IModule
    {
    private:
        /**
         * @brief Внутреннее состояние процесса переименования.
         *
         * Хранит контекст, необходимый для отслеживания последовательности
         * листов одного документа и сбора статистики.
         */
        struct RenameState
        {
            std::optional<shared::Document> previousDoc{ std::nullopt };    ///< Документ, распознанный на предыдущем шаге (для склейки листов).
            int currentPage{ 1 };       ///< Текущий номер листа многолистового документа.
            size_t successCount{ 0 };   ///< Количество успешно переименованных файлов.
            size_t errorCount{ 0 };     ///< Количество ошибок при переименовании.
        };

        shared::ModuleId m_id{ shared::ModuleId::Rename };
        std::string m_name{ "Rename scans" };   ///< Имя модуля для отображения.
        std::string m_version{ "0.0.1" };       ///< Текущая версия модуля.

        std::unique_ptr<shared::IStampRecognizer> m_recognizer; ///< Интерфейс для распознавания штампа.

        // ===================================================================
        // ГРУППА 1: Утилиты поиска файлов
        // ===================================================================

        /**
         * @brief Найти файлы с заданными расширениями в папке.
         *
         * Использует вариативные шаблоны и свёртки (fold expressions) для
         * эффективного поиска файлов сразу по нескольким расширениям.
         *
         * @tparam ARGs Типы аргументов (должны конвертироваться в std::string_view).
         * @param folder Путь к папке для поиска.
         * @param args Одно или несколько расширений (например, ".xlsx" или ".jpg", ".jpeg").
         * @return Вектор путей к найденным файлам.
         * @throws std::runtime_error Если папки не существует или не найдено ни одного файла.
         */
        template<std::convertible_to<std::string_view>... ARGs>
        static std::vector<std::filesystem::path> getFilesFromFolder(const std::filesystem::path& folder, const ARGs&... args)
        {
            if(folder.empty())
            {
                throw std::runtime_error("[pdm::modules::Rename::getFilesFromFolder] Error: folder path is empty");
            }
            if(sizeof...(args) == 0)
            {
                throw std::runtime_error("[pdm::modules::Rename::getFilesFromFolder] Error: extensions list is empty");
            }
            if(!std::filesystem::exists(folder))
            {
                throw std::runtime_error("[pdm::modules::Rename::getFilesFromFolder] Error: folder does not exist: "
                                         + folder.string());
            }

            std::vector<std::filesystem::path> foundFiles;

            for(const auto& entry : std::filesystem::directory_iterator{ folder })
            {
                if(entry.is_regular_file() && ( (args == entry.path().extension()) || ... ) )
                {
                    foundFiles.push_back(entry.path());
                }
            }

            if(foundFiles.empty())
            {
                std::string extList;
                ((extList += std::string(args) + " "), ...);
                throw std::runtime_error("[pdm::modules::Rename::getFilesFromFolder] Error: no files with extensions ["
                                         + extList
                                         + "] in folder: "
                                         + folder.string());
            }

            return foundFiles;
        }

        /**
         * @brief Найти все отсканированные изображения в папке и отсортировать их.
         *
         * @param folder Путь к папке со сканами.
         * @return Отсортированный вектор путей к найденным изображениям.
         * @throws std::runtime_error Если в папке нет изображений.
         */
        std::vector<std::filesystem::path> getScansPathsFromFolder(std::filesystem::path folder);

        // ===================================================================
        // ГРУППА 2: Основная логика переименования
        // ===================================================================

        /**
         * @brief Запустить процесс переименования для всех найденных файлов.
         *
         * @details Создает лог-файл, инициализирует состояние (RenameState) и
         * поочередно передает файлы в processSingleFile.
         *
         * @param scanFiles Список путей к сканам (отсортированный).
         * @param scansFolder Папка со сканами (используется для сохранения лога).
         */
        void performRename(const std::vector<std::filesystem::path>& scanFiles,
                           const std::filesystem::path& scansFolder);

        /**
         * @brief Обработать один файл: распознать, сформировать имя и переименовать.
         *
         * @details Если штамп не распознан, пишет ошибку в лог и сбрасывает состояние.
         * Если распознан, проверяет, является ли файл продолжением предыдущего документа.
         *
         * @param file Путь к текущему файлу.
         * @param state Ссылка на объект состояния (обновляется в процессе работы).
         * @param logger Ссылка на логгер.
         */
        void processSingleFile(const std::filesystem::path& file,
                               RenameState& state,
                               RenameLogger& logger);

        /**
         * @brief Сформировать новое имя файла на основе документа и старого пути.
         *
         * @param doc Документ (содержит номер, код, наименование и количество листов).
         * @param oldPath Старый путь к файлу (используется для извлечения расширения).
         * @param page Номер текущего листа (если листов больше 1).
         * @return Новое имя файла (без пути).
         */
        std::string buildNewName(const shared::Document& doc, const std::filesystem::path& oldPath, int page = 0);

        // ===================================================================
        // ГРУППА 3: Файловые операции
        // ===================================================================

        /**
         * @brief Безопасно переименовать один файл.
         *
         * @details Проверяет, не существует ли уже файл с целевым именем.
         * Выполняет переименование через std::filesystem. Логирует результат.
         *
         * @param oldPath Старый путь к файлу.
         * @param newPath Новый путь к файлу.
         * @param logger Ссылка на логгер для записи результата.
         * @return true, если переименование прошло успешно, иначе false.
         */
        bool renameOneFile(const std::filesystem::path& oldPath, const std::filesystem::path& newPath, RenameLogger& logger);


    public:
        // ===================================================================
        // Публичный интерфейс (Реализация контракта IModule)
        // ===================================================================

        /**
         * @brief Конструктор с внедрением зависимости (Dependency Injection).
         *
         * @param recognizer Умный указатель на распознаватель штампов (IStampRecognizer).
         */
        Rename(std::unique_ptr<shared::IStampRecognizer> recognizer)
            : m_recognizer{ std::move(recognizer) }
        {}

        ~Rename() override = default;

        /**
         * @brief Получить ID модуля.
         * @return ModuleId::Rename.
         */
        shared::ModuleId getId() const override { return m_id; }

        /**
         * @brief Получить имя модуля.
         * @return Строка с именем.
         */
        std::string_view getName() const override { return m_name; }

        /**
         * @brief Получить версию модуля.
         * @return Строка с версией.
         */
        std::string_view getVersion() const override { return m_version; }

        /**
         * @brief Запустить процесс переименования (главный метод модуля).
         * @param params Параметры задачи (должны содержать ключ scansFolder).
         */
        void execute(const shared::JobParams& params) override;
    };

    /**
     * @brief Маркер BOM (Byte Order Mark) для кодировки UTF-8.
     *
     * @details Добавляется в начало файла, чтобы Microsoft Excel
     * и другие программы корректно распознавали кириллицу (русские буквы)
     * при открытии CSV-файла.
     */
    inline constexpr const char* kUtf8Bom{ "\xEF\xBB\xBF" };
}

namespace pdm::modules
{

    /**
     * @brief Простой логгер для записи операций переименования в CSV.
     *
     * @details Открывает файл при создании и закрывает при уничтожении.
     * Записывает строки в формате: `old_name;new_name;status`.
     */
    class RenameLogger final
    {
    private:
        std::ofstream m_file;   ///< Поток для записи в файл.

    public:
        /**
         * @brief Конструктор с открытием файла лога.
         * @param path Путь к файлу лога в виде строки.
         * @throws std::runtime_error Если файл не удалось создать или открыть.
         */
        explicit RenameLogger(const std::string& path)
        {
            open(path);
        }

        /**
         * @brief Конструктор с открытием файла лога (перегрузка для std::filesystem::path).
         * @param path Путь к файлу лога.
         * @throws std::runtime_error Если файл не удалось создать или открыть.
         */
        explicit RenameLogger(const std::filesystem::path& path)
            : RenameLogger{ path.string() }
        {}




        /**
         * @brief Открыть файл лога (перегрузка для std::filesystem::path).
         * @param path Путь к файлу лога.
         */
        void open(const std::filesystem::path& path)
        {
            open(path.string());
        }

        /**
         * @brief Открыть файл лога.
         *
         * @details Открывает файл в режиме перезаписи (trunc), записывает BOM
         * для поддержки UTF-8 и заголовок таблицы.
         *
         * @param path Путь к файлу лога.
         * @throws std::runtime_error Если не удалось создать файл.
         */
        void open(const std::string& path)
        {
            m_file.open(path, std::ios::out | std::ios::trunc);

            if(!m_file.is_open())
            {
                throw std::runtime_error("RanameLogger: Error: cannot create log file: " + path);
            }

            m_file << kUtf8Bom;

            m_file << "old;new;status\n";
        }



        /**
         * @brief Записать операцию в лог.
         *
         * @param oldName Старое имя файла.
         * @param newName Новое имя файла.
         * @param status Статус операции (например, "OK" или "ERROR: ...").
         * @throws std::runtime_error Если файл лога не был открыт.
         */
        void log(const std::string& oldName, const std::string& newName, const std::string& status)
        {
            if(!m_file.is_open())
            {
                throw std::runtime_error("RanameLogger::Log: Error: Lof file is not open");
            }

            m_file << std::format("{};{};{}\n", oldName, newName, status);
            m_file.flush(); // Принудительно сбрасываем буфер на диск, чтобы не потерять данные при сбое
        }



        /**
         * @brief Проверить, открыт ли файл лога.
         * @return true, если файл открыт, иначе false.
         */
        bool isOpen() const
        {
            return m_file.is_open();
        }

    };

    inline std::string_view DocumentTypeCodeToString(DocumentTypeCode code, bool isFullCode = false)
    {
        if(!isFullCode && (code == DocumentTypeCode::DetailDrawing || code == DocumentTypeCode::Specification))
        {
            return {};
        }

        for(const auto& entry : k_documentTypeCodeMap)
        {
            if(entry.first == code)
            {
                return entry.second;
            }
        }

        return {};
    }

    /**
     * @brief Интерфейс (контракт), который обязан реализовать любой модуль системы.

     * Ядро системы (Core) работает только с этим интерфейсом и ничего не знает

    {
    public:

        virtual ~IModule() = default;

        virtual ModuleId getId() const = 0;

        virtual std::string_view getVersion() const = 0;

#ifndef JOBPARAMS_H
#define JOBPARAMS_H

#include <format>
#include <map>
#include <stdexcept>
#include <string>

namespace pdm::shared
{
    /**
     * @brief Контейнер для передачи динамических параметров в модули.
     *
     * Позволяет ядру (Core) передавать произвольный набор параметров (ключ-значение)
     * в модуль, не зная заранее, какие именно параметры нужны конкретному модулю.
     * Модули сами извлекают нужные им ключи.
     */
    class JobParams
    {
    private:
        /**
         * @brief Хранилище параметров.
         *
         * Используется std::less<> (прозрачный компаратор) для того, чтобы
         * можно было искать по std::string_view без создания временной std::string,
         * что повышает производительность.
         */
        std::map<std::string, std::string, std::less<>> m_params;

    public:
        /**
         * @brief Установить параметр.
         * @param key Ключ параметра (например, "excel_path").
         * @param value Значение параметра (например, "C:/spec.xls").
         */
        void set(std::string_view key, std::string_view value)
        {
            m_params.emplace(key, value);
        }

        /**
         * @brief Получить значение параметра по ключу.
         * @param key Ключ параметра.
         * @return Значение параметра в виде строки.
         * @throws std::runtime_error Если параметр с таким ключом не найден.
         */
        std::string get(std::string_view key) const
        {
            auto it{ m_params.find(key) };
            if(it == m_params.end())
            {
                throw std::runtime_error(std::format("Key [{}] is not found", key));
            }

            return it->second;
        }

        /**
         * @brief Проверить наличие параметра.
         * @param key Ключ параметра.
         * @return true, если параметр существует, иначе false.
         */
        bool has(std::string_view key) const
        {
            return m_params.find(key) != m_params.end();
        }
    };
} // end namespace

    /**
     * @brief Типизированный идентификатор модуля.
     *
     * Внутри ядра используется enum для безопасности и скорости (O(1) без хеширования).
     * Строки используются только на границах (CLI, JSON).
     * При добавлении нового модуля не забудьте добавить его сюда и в kAllModuleIds.
     */
    enum class ModuleId
    {
        Rename,
        NumberOfElements ///< Служебный элемент, обозначающий количество модулей (должен быть последним).
    };



    /**
     * @brief Преобразовать строковое имя модуля в типизированный ID.
     * @param s Строковое имя (например, "rename").
     * @return ModuleId, если имя найдено, иначе std::nullopt.
     */
    constexpr inline std::optional<ModuleId>  ParseModuleId(std::string_view s)
    {
        if( s == "rename")  return ModuleId::Rename;

        return std::nullopt;
    }



    /**
     * @brief Преобразовать типизированный ID в строку (для логов и UI).
     * @param id Идентификатор модуля.
     * @return Строковое имя модуля.
     * @throws std::logic_error Если передан неверный ID (например, NumberOfElements).
     */
    constexpr inline std::string_view idToString(ModuleId id)
    {
        switch(id)
        {
        case ModuleId::Rename:
            return "rename";
        case ModuleId::NumberOfElements:
            throw std::logic_error("pdm::shared::idToString: NumberOfElements is not a valid modul");
        }

        throw std::logic_error("pdm::shared::idToString: unknown ModuleId");

        return "<unknown>";
    }



    /**
     * @brief Массив всех доступных модулей.
     * @details Используется в Composition Root для автоматической регистрации.
     * При добавлении нового модуля просто добавьте его в этот массив.
     */
    inline constexpr std::array<ModuleId, static_cast<size_t>
                                (ModuleId::NumberOfElements)> kAllModuleIds{ ModuleId::Rename };

}

    pdm::shared::Document original{"716.36.15", pdm::shared::DocumentTypeCode::AssemblyDrawing, "Полка" };

    SECTION("Copy constructor")
    {
        pdm::shared::Document copy{ original };

        // Проверяем, что все поля скопированы правильно
        CHECK(copy.number == original.number);
        CHECK(copy.name   == original.name);
        CHECK(copy.code   == original.code);
    }

    SECTION("Copy assignment operator")
    {
        pdm::shared::Document copy;
        copy = original;

        CHECK(copy.number == original.number);
        CHECK(copy.name   == original.name);
        CHECK(copy.code   == original.code);
    }

    SECTION("Modifying the copy does not affect the original")
    {
        pdm::shared::Document copy{ original };
        copy.number = "321.125.765";
        copy.name   = "Кронштейн";
        copy.code   = pdm::shared::DocumentTypeCode::TechnicalSpecifications;

        REQUIRE(copy.number == "321.125.765");
        REQUIRE(copy.name   == "Кронштейн");
        REQUIRE(copy.code   == pdm::shared::DocumentTypeCode::TechnicalSpecifications);

        // Оригинал должен остаться неизменным
        CHECK(original.number == "716.36.15");
        CHECK(original.name   == "Полка");
        CHECK(original.code   == pdm::shared::DocumentTypeCode::AssemblyDrawing);
    }

    SECTION("Russian names work")
    {
        CHECK(original.name == "Полка");
        CHECK(original.name.size() == 10);  // 10 байт UTF-8 для "Полка"
    }
}
